cmake_minimum_required(VERSION 3.24)
project(zhphone LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "Windows x64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(gmod GIT_REPOSITORY https://github.com/danielga/garrysmod_common.git GIT_TAG f77a18d86f780a59ea30e4237016b05b790d4b70)
FetchContent_GetProperties(gmod)
if(NOT gmod_POPULATED)
 FetchContent_Populate(gmod)
endif()
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_MakeAvailable(minhook)
if(FFMPEG_ROOT)
 set(FFMPEG_INCLUDE_DIRS "${FFMPEG_ROOT}/include")
 foreach(component avcodec avformat avutil)
  find_library(lib_${component} NAMES ${component}-zhphone32 ${component} PATHS "${FFMPEG_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  list(APPEND FFMPEG_LIBRARIES "${lib_${component}}")
 endforeach()
else()
 find_package(FFMPEG REQUIRED)
endif()
add_library(zhphone SHARED src/module.cpp src/audio_capture.h src/phone_ear.h)
target_include_directories(zhphone PRIVATE ${gmod_SOURCE_DIR}/include ${FFMPEG_INCLUDE_DIRS})
target_link_directories(zhphone PRIVATE ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(zhphone PRIVATE minhook ${FFMPEG_LIBRARIES} d3d9 d3d11 dxgi dxguid user32 ole32 oleaut32 windowscodecs)
set_target_properties(zhphone PROPERTIES PREFIX "" OUTPUT_NAME "gmcl_zhphone_win64")
if(MSVC)
 target_compile_options(zhphone PRIVATE /W4 /EHsc /permissive-)
elseif(MINGW)
 target_link_options(zhphone PRIVATE -static)
endif()
